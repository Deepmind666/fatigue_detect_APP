extern void UpdateChannelTopSpeed(uint8_t channel, uint8_t topSpeed);

/*-----------------------------------------------
 * 函数: ProcessCommand
 * 说明: 解析串口1数据并执行命令(非阻塞轮询)
 *----------------------------------------------*/
#define CALIB_DEFAULT_WEIGHT 1000  // 校准默认重量(字节为0,0时使用)

void ProcessCommand(void)
{
    float weight;  // 仅在读重使用

    if(!Serial_GetRxFlag())
        return;

    switch(Serial_RxPacket[0])
    {
        case 0x01:   // 新协议: 配方执行(正常冰)
        {
            uint8_t w1 = Serial_RxPacket[1];
            uint8_t v1 = Serial_RxPacket[2];  // 最高速度(20~100)
            uint8_t w2 = Serial_RxPacket[3];
            uint8_t v2 = Serial_RxPacket[4];
            uint8_t w3 = Serial_RxPacket[5];
            uint8_t v3 = Serial_RxPacket[6];
            uint8_t w4 = Serial_RxPacket[7];
            uint8_t v4 = Serial_RxPacket[8];


            // 构建四通道速度表(四等分到 20)
            UpdateChannelTopSpeed(1, v1);
            UpdateChannelTopSpeed(2, v2);
            UpdateChannelTopSpeed(3, v3);
            UpdateChannelTopSpeed(4, v4);

            Serial_Printf("CMD01 W:%u %u %u %u Vtop:%u %u %u %u Pulp:%u Bag:%u Dec:%u/%u\r\n",
                          w1,w2,w3,w4, v1,v2,v3,v4,);

            BlockingGravityCompensationForFourMotors(w1, w2, w3, w4);
            break;
        }

        case 0x02:   // 新协议: 配方执行（去冰）
        {
            uint8_t w1 = Serial_RxPacket[1];
            uint8_t v1 = Serial_RxPacket[2];
            uint8_t w2 = Serial_RxPacket[3];
            uint8_t v2 = Serial_RxPacket[4];
            uint8_t w3 = Serial_RxPacket[5];
            uint8_t v3 = Serial_RxPacket[6];
            uint8_t w4 = Serial_RxPacket[7];
            uint8_t v4 = Serial_RxPacket[8];

            HX711_TareAndSave();
            SingleMotorCompensation(1, 160);  // 原逻辑保留

            UpdateChannelTopSpeed(1, v1);
            UpdateChannelTopSpeed(2, v2);
            UpdateChannelTopSpeed(3, v3);
            UpdateChannelTopSpeed(4, v4);

            Serial_Printf("CMD02 Tare+Pre W:%u %u %u %u Vtop:%u %u %u %u Pulp:%u Bag:%u Dec:%u/%u\r\n",
                          w1,w2,w3,w4, v1,v2,v3,v4,);

            BlockingGravityCompensationForFourMotors(w1, w2, w3, w4);
            break;
        }

        case 0x03:   // 控制类: 开始动作
        {
            switch(Serial_RxPacket[1])
            {
                case 0x01: // 清洗开始
                    Motor_Ctrl(1, MOTOR_STATE_FWD_FULL , 100);
                    Motor_Ctrl(2, MOTOR_STATE_FWD_FULL , 100);
                    Motor_Ctrl(3, MOTOR_STATE_REV_FULL , 100);
                    Motor_Ctrl(4, MOTOR_STATE_REV_FULL , 100);
                    Serial_Printf("Wash Start\r\n");
                    break;

                case 0x02: // 加水开始
                    // TODO: 若需要多路阀/泵控制, 在此添加
                    Motor_Ctrl(1, MOTOR_STATE_FWD_FULL , 100); // 占位: 加水时停1号
                    Serial_Printf("Water Start\r\n");
                    break;

                default:
                    Serial_Printf("CMD03 Sub Unknown:0x%02X\r\n", Serial_RxPacket[1]);
                    break;
            }
            break;
        }

        case 0x04:   // 控制类: 停止动作
        {
            switch(Serial_RxPacket[1])
            {
                case 0x01: // 清洗停止
                    Motor_Ctrl(1, MOTOR_STATE_BRAKE , 100);
                    Motor_Ctrl(2, MOTOR_STATE_BRAKE , 100);
                    Motor_Ctrl(3, MOTOR_STATE_BRAKE , 100);
                    Motor_Ctrl(4, MOTOR_STATE_BRAKE , 100);
                    Serial_Printf("Wash Stop\r\n");
                    break;

                case 0x02: // 加水停止
                    Motor_Ctrl(1, MOTOR_STATE_BRAKE , 100);
                    Serial_Printf("Water Stop\r\n");
                    break;

                default:
                    Serial_Printf("CMD04 Sub Unknown:0x%02X\r\n", Serial_RxPacket[1]);
                    break;
            }
            break;
        }

        case 0x09:   // 急停（独立指令）
        {
            Motor_Ctrl(1, MOTOR_STATE_BRAKE , 100);
            Motor_Ctrl(2, MOTOR_STATE_BRAKE , 100);
            Motor_Ctrl(3, MOTOR_STATE_BRAKE , 100);
            Motor_Ctrl(4, MOTOR_STATE_BRAKE , 100);
            Serial_Printf("Emergency Stop\r\n");
            break;
        }

        case 0x05:   // HX711 子命令
        {
            switch(Serial_RxPacket[1])
            {
                case 0x01: // 去皮
                    HX711_TareAndSave();
                    Serial_Printf("Tare OK Off=%.2f\r\n", HX711_GetOffset());
                    Serial2_Printf("Tare OK Off=%.2f\r\n", HX711_GetOffset());
                    break;

                case 0x02: // 读重量
                    weight = HX711_GetWeight(10);
                    Serial_Printf("Weight:%d g\r\n", (uint32_t)weight);
                    Serial2_Printf("Weight:%d g\r\n", (uint32_t)weight);
                    break;

                case 0x03: // 校准
                {
                    uint16_t calibrating_weight = (Serial_RxPacket[3] << 8) | Serial_RxPacket[4];
                    if(Serial_RxPacket[3] == 0 && Serial_RxPacket[4] == 0)
                        calibrating_weight = CALIB_DEFAULT_WEIGHT;
                    HX711_UpdateCalibration(calibrating_weight);
                    Serial_Printf("Cal OK %u g\r\n", calibrating_weight);
                    Serial2_Printf("Cal OK %u g\r\n", calibrating_weight);
                    break;
                }

                default:
                    Serial_Printf("HX Sub Unknown:0x%02X\r\n", Serial_RxPacket[1]);
                    break;
            }
            break;
        }

        default:
            Serial2_Printf("Unknown cmd:0x%02X\r\n", Serial_RxPacket[0]);
            break;
    }
}

/*-----------------------------------------------
 * main
 *----------------------------------------------*/
int main(void)
{
    SystemInit();
    Delay_Init();
    Serial_Init();
    Serial2_Init();
    HX711_Init();
    Motors_Init();
    DS18B20_Init();
    PA8_Init();

    Serial_Printf("System Ready\r\n");

    PA8_SetHigh();

    while(1)
    {
        ProcessCommand();
        (void)DS18B20_GetTemp(); // TODO: 如需温度输出可打印
        // TODO: 可添加非阻塞任务调度
    }
}
